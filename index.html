async function triggerAIScanWithRetry() { const apiKey = state.settings.geminiKey; if (!apiKey) return alert("Chưa có API Key!"); const pill = document.getElementById('ai-status-pill'); const laser = document.getElementById('scan-laser'); laser.classList.remove('hidden'); // Ưu tiên dòng lite để tránh nghẽn server const targetModel = "gemini-2.0-flash-lite"; const endpoint = `https://generativelanguage.googleapis.com/v1beta/models/${targetModel}:generateContent?key=${apiKey}`; const prompt = `Trích xuất tất cả số tiền thành tiền của từng món trong ảnh hóa đơn này. Xuất ra DUY NHẤT một mảng JSON các số nguyên: [35000, 75000, 120000]. Không kèm chữ hay giải thích.`; let parsed = null; let maxRetries = 3; // Tự động thử lại tối đa 3 lần nếu server bận for (let attempt = 1; attempt <= maxRetries; attempt++) { try { pill.innerText = attempt === 1 ? "Đang xử lý..." : `Server bận, thử lại lần ${attempt}...`; const response = await fetch(endpoint, { method: 'POST', headers: { 'Content-Type': 'application/json' }, body: JSON.stringify({ contents: [{ parts: [ { text: prompt }, { inline_data: { mime_type: "image/jpeg", data: state.currentBase64 } } ] }], generationConfig: { temperature: 0.1, responseMimeType: "application/json" } }) }); const resData = await response.json(); // Nếu gặp lỗi quá tải (503 hoặc 429), chờ 2 giây rồi thử lại if (response.status === 503 || response.status === 429 || resData.error?.message?.includes('high demand')) { console.warn(`Lần ${attempt} server bận. Đang chờ thử lại...`); await new Promise(resolve => setTimeout(resolve, 2000)); continue; } if (!response.ok || resData.error) { throw new Error(resData.error?.message || `Lỗi HTTP ${response.status}`); } const candidate = resData.candidates?.[0]; const textPart = candidate?.content?.parts?.find(p => !p.thought && p.text); if (textPart?.text) { parsed = JSON.parse(textPart.text.trim()); if (Array.isArray(parsed) && parsed.length > 0) break; } } catch (err) { if (attempt === maxRetries) { alert(`Không thể quét do server quá tải: ${err.message}`); } } } laser.classList.add('hidden'); if (parsed && Array.isArray(parsed)) { state.items = parsed.map((val, idx) => ({ id: Date.now() + idx, name: `Mục #${idx + 1}`, price: typeof val === 'number' ? val : (val.price || parseInt(val) || 0), consumers: state.members.map(m => m.id) })); renderItems(); calculateSettlement(); pill.innerText = `Đã lấy ${state.items.length} mục tiền 🎉`; } }
